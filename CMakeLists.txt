cmake_minimum_required(VERSION 3.20)
project(orbiSteamworks VERSION 0.0.1 LANGUAGES C CXX)
set(OSW_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(OSW_STANDALONE ON)
endif()
option(OSW_BUILD_TOOLS "Build authentication programs" ${OSW_STANDALONE})
option(OSW_BUILD_TESTS "Build orbiSteamworks contract tests" ${OSW_STANDALONE})
option(OSW_BUILD_SHARED "Build the runtime-loadable orbisteamworks shared library" OFF)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The runtime loader currently supports 64-bit hosts only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET nlohmann_json::nlohmann_json)
  find_package(nlohmann_json 3.11 QUIET)
  if(NOT TARGET nlohmann_json::nlohmann_json)
    find_path(OSW_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
    add_library(osw_json INTERFACE)
    target_include_directories(osw_json INTERFACE "${OSW_JSON_INCLUDE_DIR}")
    add_library(nlohmann_json::nlohmann_json ALIAS osw_json)
  endif()
endif()
add_library(osw_credential STATIC src/core/credential.cpp)
target_include_directories(osw_credential PUBLIC src)
add_library(osw_steam STATIC src/steam/runtime.cpp)
target_include_directories(osw_steam PUBLIC src)
target_link_libraries(osw_steam PRIVATE ${CMAKE_DL_LIBS})
set(OSW_LIBRARY_SOURCES src/api.cpp src/steam/backend.cpp src/core/platform.cpp src/core/multipart.cpp src/core/blobs.cpp)
add_library(orbisteamworks STATIC ${OSW_LIBRARY_SOURCES})
target_include_directories(orbisteamworks PUBLIC include PRIVATE src)
target_link_libraries(orbisteamworks PRIVATE osw_steam osw_credential nlohmann_json::nlohmann_json)
set(OSW_TARGETS orbisteamworks osw_credential osw_steam)
if(OSW_BUILD_SHARED)
  # Compiled separately from the static library so that the plugin's runtime
  # library and symbol visibility never change what in-process hosts link.
  # Only the osw_* C functions are exported (see docs/PLUGIN_DESIGN.md).
  find_package(Threads REQUIRED)
  add_library(orbisteamworks_shared SHARED ${OSW_LIBRARY_SOURCES} src/core/credential.cpp src/steam/runtime.cpp)
  target_include_directories(orbisteamworks_shared PUBLIC include PRIVATE src)
  target_link_libraries(orbisteamworks_shared PRIVATE nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
  set_target_properties(orbisteamworks_shared PROPERTIES OUTPUT_NAME orbisteamworks
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(WIN32)
    target_sources(orbisteamworks_shared PRIVATE src/orbisteamworks.def)
    # Hosts load the DLL at runtime; keep the unused import library apart from
    # the static orbisteamworks.lib.
    set_target_properties(orbisteamworks_shared PROPERTIES ARCHIVE_OUTPUT_NAME orbisteamworks_shared)
    if(MSVC)
      # No CRT object crosses the C API, so the DLL needs no VC++ redistributable.
      set_target_properties(orbisteamworks_shared PROPERTIES
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(OSW_VERSION_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/src/orbisteamworks.map")
    target_link_options(orbisteamworks_shared PRIVATE
      "LINKER:--version-script=${OSW_VERSION_SCRIPT}" "LINKER:--no-undefined")
    set_property(TARGET orbisteamworks_shared APPEND PROPERTY LINK_DEPENDS "${OSW_VERSION_SCRIPT}")
  else()
    message(FATAL_ERROR "OSW_BUILD_SHARED supports Windows and Linux hosts only")
  endif()
  list(APPEND OSW_TARGETS orbisteamworks_shared)
endif()
if(OSW_BUILD_TOOLS OR OSW_BUILD_TESTS)
add_library(osw_spike_common STATIC tools/auth-spike/common.cpp)
target_include_directories(osw_spike_common PUBLIC tools/auth-spike)
target_link_libraries(osw_spike_common PUBLIC osw_steam osw_credential)
list(APPEND OSW_TARGETS osw_spike_common)
endif()
if(OSW_BUILD_TOOLS)
add_executable(osw-auth-client tools/auth-spike/client/main.cpp)
add_executable(osw-auth-server tools/auth-spike/server/main.cpp)
add_executable(osw-auth-helper server/auth-helper/main.cpp)
add_executable(osw-platform-check tools/platform-check/main.cpp)
target_link_libraries(osw-platform-check PRIVATE orbisteamworks osw_spike_common nlohmann_json::nlohmann_json)
target_link_libraries(osw-auth-client PRIVATE osw_spike_common)
target_link_libraries(osw-auth-server PRIVATE osw_spike_common)
target_link_libraries(osw-auth-helper PRIVATE osw_spike_common)
list(APPEND OSW_TARGETS osw-auth-client osw-auth-server osw-auth-helper osw-platform-check)
endif()
foreach(target ${OSW_TARGETS})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(OSW_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
